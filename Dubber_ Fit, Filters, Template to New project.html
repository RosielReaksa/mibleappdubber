API changes

Add these to POST /projects and POST /projects/{id}/exports. Templates are plain settings, so GET /templates returns the same fields.

"fit": { "mode": "blur", "blur_strength": 60 },   // blur | bars | fill
"filter": { "name": "warm", "strength": 70 },     // none | warm | cool | mono | vivid | soft

// GET /templates item
{ "id": "tpl_short", "name": "Short video",
  "settings": { "format": "portrait", "subtitles": {"show": true, "style": "bold"},
    "mix": {"dubbed_voice": 100, "original_voice": 15, "music": 80},
    "fit": {"mode": "blur", "blur_strength": 60},
    "logo_position": "top_right" } }
